impl Launcher {
    /// Active download card with a cancel button. Returns true on cancel.
    fn download_card(&mut self, ui: &mut Ui, skin: &Skin) -> bool {
        let mut cancel = false;
        let active = self
            .mod_task
            .as_ref()
            .map(|t| !t.finished)
            .unwrap_or(false);
        if active {
            let (title, frac, percent) = {
                let t = self.mod_task.as_ref().unwrap();
                (t.title.clone(), t.frac(), t.total > 0)
            };
            let frame = theme::card_frame(skin);
            frame.show(ui, |ui| {
                ui.set_width(ui.available_width() - 32.0);
                ui.horizontal(|ui| {
                    ui.label(
                        RichText::new(format!("Скачивание · {}", truncate(&title, 40)))
                            .font(theme::body(12.5))
                            .color(skin.dim()),
                    );
                    ui.with_layout(Layout::right_to_left(Align::Center), |ui| {
                        if widgets::button_sized(
                            ui,
                            skin,
                            "Отменить",
                            Btn::Ghost,
                            true,
                            24.0,
                            10.5,
                            9.5,
                        )
                        .clicked()
                        {
                            cancel = true;
                        }
                        let label = if percent {
                            format!("{}%", (frac * 100.0).round() as u32)
                        } else {
                            "…".to_string()
                        };
                        ui.label(
                            RichText::new(label)
                                .font(theme::mono(10.0))
                                .color(skin.faint()),
                        );
                    });
                });
                ui.add_space(6.0);
                widgets::progress_bar(ui, skin, frac);
            });
            ui.add_space(10.0);
        }
        cancel
    }

    pub fn view_mods(&mut self, ui: &mut Ui, skin: &Skin) {
        let (mc, loader) = self.target_version();
        let installed = self.mods_cache.clone();
        let kind_idx = self.search.kind;
        let kind = mods::KINDS[kind_idx.min(mods::KINDS.len() - 1)].0;
        let source = self.search.source;
        let page = self.mods_page;
        let busy = self.search.busy;
        let download_busy = self
            .mod_task
            .as_ref()
            .map(|t| !t.finished)
            .unwrap_or(false);
        let build_busy = self
            .build_task
            .as_ref()
            .map(|t| !t.finished)
            .unwrap_or(false);
        let mut search_now = false;
        let mut install_hit: Option<ModHit> = None;
        let mut toggle_req: Option<(String, bool)> = None;
        let mut delete_req: Option<String> = None;
        let mut confirm = self.confirm_delete.clone();
        let mut open_content = false;
        let mut add_build: Option<usize> = None;
        let mut remove_build: Option<usize> = None;
        let mut install_build = false;
        let mut clear_build = false;

        widgets::header(
            ui,
            skin,
            "Моды",
            "Каталог контента и свои сборки для Мадрида",
            |ui| {
                if widgets::button(ui, skin, "Папка игры", Btn::Ghost, true).clicked() {
                    open_content = true;
                }
            },
        );

        // page switch
        ui.horizontal(|ui| {
            if widgets::chip(ui, skin, "КАТАЛОГ", page == 0).clicked() {
                self.mods_page = 0;
            }
            let installed_n = installed.len();
            if widgets::chip(
                ui,
                skin,
                &format!("УСТАНОВЛЕННЫЕ · {installed_n}"),
                page == 1,
            )
            .clicked()
            {
                self.mods_page = 1;
            }
        });
        ui.add_space(10.0);

        if page == 0 {
            // ------------------------------------------------------ catalog
            ui.horizontal(|ui| {
                for (i, (_, label)) in mods::KINDS.iter().enumerate() {
                    if widgets::chip(ui, skin, label, kind_idx == i).clicked() {
                        self.search.kind = i;
                        self.search.searched = false;
                        self.search.results.clear();
                    }
                }
            });
            ui.add_space(8.0);
            ui.horizontal(|ui| {
                let row_w = ui.available_width();
                let field_w = (row_w - 260.0).max(160.0);
                if widgets::chip(ui, skin, "Modrinth", source == 1).clicked() {
                    self.search.source = 1;
                }
                if widgets::chip(ui, skin, "CurseForge", source == 0).clicked() {
                    self.search.source = 0;
                }
                if widgets::chip(ui, skin, "↓ скачивания", self.search.sort == 0).clicked() {
                    self.search.sort = 0;
                }
                if widgets::chip(ui, skin, "♥ популярность", self.search.sort == 1).clicked() {
                    self.search.sort = 1;
                }
                if widgets::chip(ui, skin, "↑ обновления", self.search.sort == 2).clicked() {
                    self.search.sort = 2;
                }
                if widgets::chip(ui, skin, "★ новые", self.search.sort == 3).clicked() {
                    self.search.sort = 3;
                }
                let resp = search_field(
                    ui,
                    skin,
                    "mod-search",
                    &mut self.search.query,
                    "Название",
                    field_w,
                );
                if resp.lost_focus() && ui.input(|i| i.key_pressed(egui::Key::Enter)) {
                    search_now = true;
                }
                let label = if busy { "Ищу…" } else { "Найти" };
                if widgets::button(ui, skin, label, Btn::Primary, !busy).clicked() {
                    search_now = true;
                }
            });
            ui.add_space(4.0);
            ui.horizontal(|ui| {
                ui.label(
                    RichText::new(match kind {
                        mods::KIND_MAP => {
                            if mc.is_empty() { "версия не выбрана".into() } else {
                                format!("карты живут на CurseForge — нужен ключ в настройках · цель {mc}")
                            }
                        }
                        mods::KIND_MODPACK => {
                            if mc.is_empty() { "версия не выбрана".into() } else {
                                format!("модпак встанет в текущий профиль Мадрид · цель {mc}")
                            }
                        }
                        mods::KIND_SHADER => "шейдеры идут в shaderpacks · Modrinth, ключ не нужен".to_string(),
                        mods::KIND_RESOURCEPACK => "ресурспаки идут в resourcepacks · Modrinth".to_string(),
                        _ => {
                            if mc.is_empty() { "версия не выбрана".to_string() } else {
                                format!("цель: Minecraft {mc} · {loader}")
                            }
                        }
                    })
                    .font(theme::mono(9.0))
                    .color(skin.faint()),
                );
            });
            ui.add_space(10.0);

            if let Some(err) = self.search.error.clone() {
                let frame = theme::card_frame(skin);
                frame.show(ui, |ui| {
                    ui.set_width(ui.available_width() - 32.0);
                    ui.label(
                        RichText::new(truncate(&err, 240))
                            .font(theme::body(12.0))
                            .color(skin.dim()),
                    );
                });
                ui.add_space(10.0);
            }

            if self.download_card(ui, skin) {
                self.cancel_active_downloads();
            }

            egui::ScrollArea::vertical()
                .id_salt("mods-catalog")
                .auto_shrink([false, false])
                .show(ui, |ui| {
                    ui.set_width(ui.available_width());
                    widgets::section_label(ui, skin, "РЕЗУЛЬТАТЫ");
                    ui.add_space(4.0);
                    if self.search.results.is_empty() {
                        if self.search.searched && !busy {
                            ui.add_space(8.0);
                            ui.label(
                                RichText::new("Ничего не нашлось — попробуйте другой запрос.")
                                    .font(theme::body(12.0))
                                    .color(skin.faint()),
                            );
                        } else {
                            ui.add_space(8.0);
                            ui.label(
                                RichText::new("Введите название и нажмите «Найти».")
                                    .font(theme::body(12.0))
                                    .color(skin.faint()),
                            );
                        }
                    } else {
                        let results = self.search.results.clone();
                        for (i, hit) in results.iter().enumerate() {
                            let icon = hit
                                .icon_url
                                .as_deref()
                                .and_then(|u| self.mod_icon(ui.ctx(), u));
                            let (added, do_install) = hit_row(
                                ui,
                                skin,
                                hit,
                                icon.as_ref(),
                                busy || download_busy,
                            );
                            if added {
                                add_build = Some(i);
                            }
                            if do_install {
                                install_hit = Some(hit.clone());
                            }
                        }
                    }
                    ui.add_space(12.0);
                });
        } else {
            // -------------------------------------------------- installed
            if self.download_card(ui, skin) {
                self.cancel_active_downloads();
            }

            // build panel
            let build_count = self.build.len();
            if build_count > 0 || build_busy {
                let frame = theme::card_frame(skin);
                frame.show(ui, |ui| {
                    ui.set_width(ui.available_width() - 32.0);
                    ui.horizontal(|ui| {
                        widgets::badge(ui, skin, &format!("СБОРКА · {}", build_count), true);
                        let building = self
                            .build_task
                            .as_ref()
                            .map(|t| !t.finished)
                            .unwrap_or(false);
                        let progress = self
                            .build_task
                            .as_ref()
                            .map(|t| (t.done, t.total, t.frac(), t.error.clone()));
                        if building {
                            if let Some((done, total, _, _)) = &progress {
                                ui.label(
                                    RichText::new(format!("Ставлю сборку · {done}/{total}"))
                                        .font(theme::mono(10.0))
                                        .color(skin.dim()),
                                );
                            }
                            if widgets::button_sized(
                                ui,
                                skin,
                                "Отменить",
                                Btn::Ghost,
                                true,
                                24.0,
                                10.5,
                                9.5,
                            )
                            .clicked()
                            {
                                self.cancel_active_downloads();
                            }
                        }
                        ui.with_layout(Layout::right_to_left(Align::Center), |ui| {
                            if widgets::button_sized(
                                ui,
                                skin,
                                "Установить сборку",
                                Btn::Primary,
                                !build_busy && build_count > 0,
                                28.0,
                                11.5,
                                11.0,
                            )
                            .clicked()
                            {
                                install_build = true;
                            }
                            if widgets::button_sized(
                                ui,
                                skin,
                                "Очистить",
                                Btn::Ghost,
                                !build_busy && build_count > 0,
                                28.0,
                                11.0,
                                9.5,
                            )
                            .clicked()
                            {
                                clear_build = true;
                            }
                        });
                    });
                    if build_count > 0 {
                        ui.add_space(6.0);
                        ui.horizontal_wrapped(|ui| {
                            ui.spacing_mut().item_spacing = Vec2::new(5.0, 5.0);
                            for (i, e) in self.build.clone().iter().enumerate() {
                                let label = format!("{} ✕", truncate(&e.title, 26));
                                if widgets::chip(ui, skin, &label, false).clicked() {
                                    remove_build = Some(i);
                                }
                            }
                        });
                    }
                    let building = self
                        .build_task
                        .as_ref()
                        .map(|t| !t.finished)
                        .unwrap_or(false);
                    if building {
                        let frac = self
                            .build_task
                            .as_ref()
                            .map(|t| t.frac())
                            .unwrap_or(0.0);
                        ui.add_space(6.0);
                        widgets::progress_bar(ui, skin, frac);
                        if let Some(err) = self
                            .build_task
                            .as_ref()
                            .and_then(|t| t.error.clone())
                        {
                            ui.add_space(4.0);
                            ui.label(
                                RichText::new(truncate(&err, 200))
                                    .font(theme::body(11.5))
                                    .color(skin.ink()),
                            );
                        }
                    }
                });
                ui.add_space(10.0);
            }

            egui::ScrollArea::vertical()
                .id_salt("mods-installed")
                .auto_shrink([false, false])
                .show(ui, |ui| {
                    ui.set_width(ui.available_width());
                    widgets::section_label(ui, skin, "УСТАНОВЛЕННЫЕ МОДЫ");
                    ui.add_space(4.0);
                    if installed.is_empty() {
                        widgets::empty_state(
                            ui,
                            skin,
                            Icon::Plus,
                            "Пока пусто",
                            "Открой «Каталог» и добавь моды — они лягут в папку mods.",
                        );
                    } else {
                        for m in &installed {
                            let (toggle, delete) = mod_row(
                                ui,
                                skin,
                                &m.name,
                                &m.file,
                                m.size,
                                m.enabled,
                                confirm.as_deref() == Some(m.file.as_str()),
                            );
                            if toggle {
                                toggle_req = Some((m.file.clone(), !m.enabled));
                            }
                            if delete {
                                if confirm.as_deref() == Some(m.file.as_str()) {
                                    delete_req = Some(m.file.clone());
                                    confirm = None;
                                } else {
                                    confirm = Some(m.file.clone());
                                }
                            }
                        }
                    }
                    ui.add_space(12.0);
                });
        }

        self.confirm_delete = confirm;

        if open_content {
            let dir = if self.cfg.game_dir.trim().is_empty() {
                crate::config::data_dir().join("game")
            } else {
                std::path::PathBuf::from(self.cfg.game_dir.trim())
            };
            let _ = std::fs::create_dir_all(&dir);
            crate::util::open_path(&dir);
        }
        if search_now {
            self.start_search(ui.ctx());
        }
        if let Some((file, enabled)) = toggle_req {
            match mods::set_enabled(&self.cfg, &file, enabled) {
                Ok(_) => self.toast(if enabled { "Включено" } else { "Отключено" }),
                Err(e) => self.toast(format!("{e:#}")),
            }
        }
        if let Some(file) = delete_req {
            match mods::delete(&self.cfg, &file) {
                Ok(_) => self.toast("Удалено"),
                Err(e) => self.toast(format!("{e:#}")),
            }
        }
        if let Some(hit) = install_hit {
            self.start_mod_install(ui.ctx(), hit);
        }
        if let Some(i) = add_build {
            if let Some(hit) = self.search.results.clone().get(i) {
                self.build_add(hit, mc.clone(), loader.clone());
            }
        }
        if let Some(i) = remove_build {
            self.build_remove(i);
        }
        if clear_build {
            self.build.clear();
            mods::save_build(&self.build);
            self.toast("Сборка очищена");
        }
        if install_build {
            self.start_build_install(ui.ctx());
        }
        let _ = Tab::Mods;
    }
}
